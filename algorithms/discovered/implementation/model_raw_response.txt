        self.B_k = rng.normal(0, 0.1, size=(self.K, self.d_z, self.d_u))

                self._adam_update('b_alpha', d_b_alpha)

import numpy as np


class CandidateModel:
    def __init__(self, latent_dim: int, seed: int = 0):
        self.d_z = latent_dim
        self.K = 4
        self.d_mu = 1
        self.d_u = 1
        
        self.rho = 0.95
        self.lambda_A = 1e-4
        self.lambda_B = 1e-4
        self.lambda_gate = 1e-4
        self.lambda_spec = 0.01
        self.learning_rate = 1e-3
        self.max_epochs = 200
        self.batch_size = min(32, max(1, latent_dim * 2))
        self.patience = 30
        
        rng = np.random.default_rng(seed)
        
        self.A_k = rng.normal(0, 0.05, size=(self.K, self.d_z, self.d_z))
        self.B_k = rng.normal(0, 0.05, size=(self.K, self.d_z, self.d_u))
        self.W_alpha = rng.normal(0, 0.05, size=(self.K, self.d_mu))
        self.b_alpha = np.zeros(self.K)
        self.c_A = np.zeros(self.d_z)
        self.c_B = np.zeros((self.d_z, self.d_u))
        
        self.z_mean = None
        self.z_std = None
        
        self.t_step = 0
        self.beta1 = 0.9
        self.beta2 = 0.999
        self.epsilon = 1e-8
        
        self._param_names = ['A_k', 'B_k', 'W_alpha', 'b_alpha', 'c_A', 'c_B']
        self._m = {n: np.zeros_like(getattr(self, n)) for n in self._param_names}
        self._v = {n: np.zeros_like(getattr(self, n)) for n in self._param_names}
    
    def _softmax(self, x):
        x_max = np.max(x, axis=-1, keepdims=True)
        exp_x = np.exp(x - x_max)
        return exp_x / np.sum(exp_x, axis=-1, keepdims=True)
    
    def _alpha_from_mu(self, mu):
        single = mu.ndim == 1
        if single:
            mu = mu[np.newaxis, :]
        logits = mu @ self.W_alpha.T + self.b_alpha
        alpha = self._softmax(logits)
        if single:
            alpha = alpha[0]
        return alpha
    
    def _A_eff_from_alpha(self, alpha):
        single = alpha.ndim == 1
        if single:
            alpha = alpha[np.newaxis, :]
        A_eff = np.einsum('bk,kij->bij', alpha, self.A_k)
        if single:
            A_eff = A_eff[0]
        return A_eff
    
    def _B_eff_from_alpha(self, alpha):
        single = alpha.ndim == 1
        if single:
            alpha = alpha[np.newaxis, :]
        B_eff = np.einsum('bk,kij->bij', alpha, self.B_k)
        if single:
            B_eff = B_eff[0]
        return B_eff
    
    def _svd_max(self, A):
        return np.linalg.svd(A, compute_uv=False)[0]
    
    def _svd_max_with_vecs(self, A):
        U, S, Vt = np.linalg.svd(A)
        return S[0], U[:, 0], Vt[0, :]
    
    def _adam_update(self, name, grad):
        if not np.any(np.isfinite(grad)):
            return
        self.t_step += 1
        t = self.t_step
        self._m[name] = self.beta1 * self._m[name] + (1 - self.beta1) * grad
        self._v[name] = self.beta2 * self._v[name] + (1 - self.beta2) * (grad ** 2)
        m_hat = self._m[name] / (1 - self.beta1 ** t)
        v_hat = self._v[name] / (1 - self.beta2 ** t)
        param = getattr(self, name)
        update = self.learning_rate * m_hat / (np.sqrt(v_hat) + self.epsilon)
        if not np.all(np.isfinite(update)):
            return
        param -= update
        setattr(self, name, param)
    
    def _predict_from_norm(self, z_t_norm):
        alpha = self._alpha_from_mu(np.zeros(self.d_mu))
        A_eff = self._A_eff_from_alpha(alpha)
        z_next_norm = A_eff @ z_t_norm + self.c_A
        return z_next_norm
    
    def fit(self, z_train):

        T = z_train.shape[0]
        
        if T < 2:

                sv = self._svd_max(A_eff_init)
                if sv > self.rho:
                    U, S, Vt = np.linalg.svd(A_eff_init)
                    S = np.minimum(S, self.rho)
                    A_eff_init = U @ np.diag(S) @ Vt
                
                # Distribute across A_k
                for k in range(self.K):

                self.c_A = c_A_init
            except np.linalg.LinAlgError:

        z_t_norm = (z_t - self.z_mean) / self.z_std
        z_next_norm = self._predict_from_norm(z_t_norm)
        z_next = z_next_norm * self.z_std + self.z_mean
        return z_next

import numpy as np


class CandidateModel:
    def __init__(self, latent_dim: int, seed: int = 0):
        self.d_z = latent_dim
        self.K = 4
        self.d_mu = 1
        self.d_u = 1
        
        self.rho = 0.95
        self.lambda_A = 1e-4
        self.lambda_B = 1e-4
        self.lambda_gate = 1e-4
        self.lambda_spec = 0.01
        self.learning_rate = 1e-3
        self.max_epochs = 200
        self.batch_size = min(32, max(1, latent_dim * 2))
        self.patience = 30
        
        rng = np.random.default_rng(seed)
        
        self.A_k = rng.normal(0, 0.05, size=(self.K, self.d_z, self.d_z))
        self.B_k = rng.normal(0, 0.05, size=(self.K, self.d_z, self.d_u))
        self.W_alpha = rng.normal(0, 0.05, size=(self.K, self.d_mu))
        self.b_alpha = np.zeros(self.K)
        self.c_A = np.zeros(self.d_z)
        self.c_B = np.zeros((self.d_z, self.d_u))
        
        self.z_mean = None
        self.z_std = None
        
        self.t_step = 0
        self.beta1 = 0.9
        self.beta2 = 0.999
        self.epsilon = 1e-8
        
        self._param_names = ['A_k', 'B_k', 'W_alpha', 'b_alpha', 'c_A', 'c_B']
        self._m = {n: np.zeros_like(getattr(self, n)) for n in self._param_names}
        self._v = {n: np.zeros_like(getattr(self, n)) for n in self._param_names}
    
    def _softmax(self, x):
        x_max = np.max(x, axis=-1, keepdims=True)
        exp_x = np.exp(x - x_max)
        return exp_x / np.sum(exp_x, axis=-1, keepdims=True)
    
    def _alpha_from_mu(self, mu):
        single = mu.ndim == 1
        if single:
            mu = mu[np.newaxis, :]
        logits = mu @ self.W_alpha.T + self.b_alpha
        alpha = self._softmax(logits)
        if single:
            alpha = alpha[0]
        return alpha
    
    def _A_eff_from_alpha(self, alpha):
        single = alpha.ndim == 1
        if single:
            alpha = alpha[np.newaxis, :]
        A_eff = np.einsum('bk,kij->bij', alpha, self.A_k)
        if single:
            A_eff = A_eff[0]
        return A_eff
    
    def _svd_max(self, A):
        return np.linalg.svd(A, compute_uv=False)[0]
    
    def _svd_max_with_vecs(self, A):
        U, S, Vt = np.linalg.svd(A)
        return S[0], U[:, 0], Vt[0, :]
    
    def _adam_update(self, name, grad):
        if not np.any(np.isfinite(grad)):
            return
        self.t_step += 1
        t = self.t_step
        self._m[name] = self.beta1 * self._m[name] + (1 - self.beta1) * grad
        self._v[name] = self.beta2 * self._v[name] + (1 - self.beta2) * (grad ** 2)
        m_hat = self._m[name] / (1 - self.beta1 ** t)
        v_hat = self._v[name] / (1 - self.beta2 ** t)

        update = self.learning_rate * m_hat / (np.sqrt(v_hat) + self.epsilon)
        if not np.all(np.isfinite(update)):
            return
        param -= update
        setattr(self, name, param)

        z_next_norm = A_eff @ z_t_norm + self.c_A

        T = z_train.shape[0]

            return

        self.z_std = np.std(z_train, axis=0) + 1e-8

        z_t_all = z_norm[:-1]
        z_next_all = z_norm[1:]

                theta = z_next_all.T @ X @ np.linalg.inv(XtX)

                c_A_init = theta[:, self.d_z]

                if sv > self.rho:
                    U, S, Vt = np.linalg.svd(A_eff_init)
                    S = np.minimum(S, self.rho)
                    A_eff_init = U @ np.diag(S) @ Vt

                    self.A_k[k] = A_eff_init
                self.c_A = c_A_init
            except np.linalg.LinAlgError:
                pass
        
        val_start = max(1, int(0.8 * N))
        
        best_val = np.inf
        patience_counter = 0
        best_params = None
        
        for epoch in range(self.max_epochs):
            train_idx = np.random.permutation(val_start)
            epoch_loss = 0.0
            n_batches = 0
            
            for start in range(0, len(train_idx), self.batch_size):
                end = min(start + self.batch_size, len(train_idx))
                batch_idx = train_idx[start:end]
                if len(batch_idx) == 0:
                    continue
                
                z_t_batch = z_t_all[batch_idx]
                z_next_batch = z_next_all[batch_idx]
                B = z_t_batch.shape[0]
                
                alpha = self._alpha_from_mu(np.zeros(self.d_mu))
                A_eff = self._A_eff_from_alpha(alpha)
                
                z_pred = z_t_batch @ A_eff.T + self.c_A
                
                one_step_loss = np.mean((z_pred - z_next_batch) ** 2)
                reg_loss = (self.lambda_A * np.sum(self.A_k ** 2) +
                           self.lambda_B * np.sum(self.B_k ** 2) +
                           self.lambda_gate * (np.sum(self.W_alpha ** 2) + 
                                               np.sum(self.b_alpha ** 2) +
                                               np.sum(self.c_A ** 2)))
                sv = self._svd_max(A_eff)
                spec_loss = self.lambda_spec * max(0.0, sv - self.rho)
                
                loss = one_step_loss + reg_loss + spec_loss
                
                if not np.isfinite(loss):
                    continue
                
                epoch_loss += loss
                n_batches += 1

                d_A_eff = d_z_pred.T @ z_t_batch

                d_A_k = alpha[:, None, None] * d_A_eff[None, :, :]
                
                d_A_k += 2 * self.lambda_A * self.A_k
                d_c_A += 2 * self.lambda_gate * self.c_A
                
                if sv > self.rho:
                    _, u_max, v_max = self._svd_max_with_vecs(A_eff)
                    d_A_eff_spec = self.lambda_spec * np.outer(u_max, v_max)
                    d_A_k += alpha[:, None, None] * d_A_eff_spec[None, :, :]
                
                d_W_alpha = 2 * self.lambda_gate * self.W_alpha
                d_b_alpha = 2 * self.lambda_gate * self.b_alpha
                
                self._adam_update('A_k', d_A_k)
                self._adam_update('W_alpha', d_W_alpha)
                self._adam_update('b_alpha', d_b_alpha)
                self._adam_update('c_A', d_c_A)
            
            if val_start < N:
                z_t_val = z_t_all[val_start:]
                z_next_val = z_next_all[val_start:]
                alpha_val = self._alpha_from_mu(np.zeros(self.d_mu))
                A_eff_val = self._A_eff_from_alpha(alpha_val)
                z_pred_val = z_t_val @ A_eff_val.T + self.c_A
                val_loss = np.mean((z_pred_val - z_next_val) ** 2)
            else:
                val_loss = epoch_loss / max(n_batches, 1)
            
            if not np.isfinite(val_loss):
                val_loss = np.inf
            
            if val_loss < best_val:
                best_val = val_loss
                patience_counter = 0
                best_params = {n: getattr(self, n).copy() for n in self._param_names}
            else:
                patience_counter += 1
                if patience_counter >= self.patience:
                    break
        
        if best_params is not None:
            for n, p in best_params.items():
                setattr(self, n, p)
    
    def predict_next(self, z_t):
        z_t = np.asarray(z_t, dtype=np.float64)
        z_t_norm = (z_t - self.z_mean) / self.z_std
        z_next_norm = self._predict_from_norm(z_t_norm)
        z_next = z_next_norm * self.z_std + self.z_mean
        return z_next
    
    def rollout(self, z0, horizon):
        z0 = np.asarray(z0, dtype=np.float64)
        rollout = np.zeros((horizon + 1, self.d_z))
        rollout[0] = z0.copy()
        for i in range(horizon):
            rollout[i+1] = self.predict_next(rollout[i])
        return rollout
